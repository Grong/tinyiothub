- [ ] **Step 2: 重构 Manager 为单引擎**

核心循环（替换 per-owner start_push spawn）：

```rust
// derived_manager.rs：start() 末尾 spawn 唯一订阅任务
let mut rx = self.events.subscribe();
let index = self.index.clone();       // Arc<RwLock<HashMap<(String,String), Vec<String>>>>
let defs_map = self.defs.clone();     // Arc<RwLock<HashMap<owner_id, DerivedDefsHandle>>>
let cache = self.cache.clone();
let sink = self.sink.clone();
let status = self.status.clone();
tokio::spawn(async move {
    loop {
        match rx.recv().await {
            Ok(event) => {
                if event.event_type() != ThingEventType::PropertyChange { continue; }
                let src = event.source();
                // 命中反向索引（thing_id 级）
                let owners = index.read().get(&(src.source_id.clone(), String::new()))
                    .cloned()
                    .unwrap_or_default();
                // 索引键语义：(ref_thing_id, ref_property_name)；property 过滤在 def 级
                for owner in owners {
                    let Some(handle) = defs_map.read().get(&owner).cloned() else { continue };
                    let defs = handle.read().clone();
                    // 仅重算引用该 (thing,property) 的 defs（含 property 级过滤）
                    let values = recompute_defs(&defs, Some(&src), &cache, &counter).await;
                    if !values.is_empty() {
                        let _ = sink.send(DriverReading { thing_id: owner.clone(), values }).await;
                    }
                }
            }
            Err(RecvError::Lagged(n)) => {
                DERIVED_EVENTBUS_LAGGED.fetch_add(n, Ordering::Relaxed);
                // 全量重算自愈（语义保持）
            }
            Err(RecvError::Closed) => break,
        }
    }
});
```

注：反向索引键保留 `(thing_id, property_name)`；命中后 def 级再按 refs 精确过滤（消除 Task 7 台账"只比 thing_id 过触发"的 Minor——顺手收编）。`recompute_defs` 由 derived.rs 的 recompute 改造（async，descendant_count 经 port 预取，Task 7 抉择①语义保持）。
